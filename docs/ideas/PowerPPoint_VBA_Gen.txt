Sub StyleTitleSlide(sld As Slide, titleText As String, subText As String)
    ' Deep Navy Background
    sld.FollowMasterBackground = msoFalse
    sld.Background.Fill.ForeColor.RGB = RGB(15, 32, 64)
    
    ' Title Text
    Dim shpTitle As Shape
    Set shpTitle = sld.Shapes.AddTextbox(msoTextOrientationHorizontal, 50, 180, 860, 100)
    With shpTitle.TextFrame.TextRange
        .Text = titleText
        .Font.Name = "Arial"
        .Font.Size = 60
        .Font.Bold = msoTrue
        .Font.Color.RGB = RGB(214, 245, 232) ' Mint Green
        .ParagraphFormat.Alignment = ppAlignCenter
    End With
    
    ' Subtitle Text
    Dim shpSub As Shape
    Set shpSub = sld.Shapes.AddTextbox(msoTextOrientationHorizontal, 100, 280, 760, 100)
    With shpSub.TextFrame.TextRange
        .Text = subText
        .Font.Name = "Arial"
        .Font.Size = 24
        .Font.Color.RGB = RGB(255, 255, 255) ' White
        .ParagraphFormat.Alignment = ppAlignCenter
    End With
End Sub

Sub StyleContentSlide(sld As Slide, titleText As String, bodyText As String)
    ' Light Gray/Off-White Web Background
    sld.FollowMasterBackground = msoFalse
    sld.Background.Fill.ForeColor.RGB = RGB(252, 251, 249)
    
    ' Deep Navy Title
    Dim shpTitle As Shape
    Set shpTitle = sld.Shapes.AddTextbox(msoTextOrientationHorizontal, 50, 30, 860, 60)
    With shpTitle.TextFrame.TextRange
        .Text = titleText
        .Font.Name = "Arial"
        .Font.Size = 32
        .Font.Bold = msoTrue
        .Font.Color.RGB = RGB(15, 32, 64)
    End With
    
    ' White Rounded Card with Soft Shadow
    Dim shpCard As Shape
    Set shpCard = sld.Shapes.AddShape(msoShapeRoundedRectangle, 50, 110, 860, 380)
    With shpCard
        .Fill.ForeColor.RGB = RGB(255, 255, 255)
        .Line.Visible = msoFalse
        .Adjustments.Item(1) = 0.04 ' Corner radius
        .Shadow.Type = msoShadow21 ' Soft drop shadow
        .Shadow.Transparency = 0.8
        .Shadow.Size = 100
        .Shadow.Blur = 15
    End With
    
    ' Body Text Inside Card
    Dim shpBody As Shape
    Set shpBody = sld.Shapes.AddTextbox(msoTextOrientationHorizontal, 80, 140, 800, 320)
    With shpBody.TextFrame
        .WordWrap = msoTrue
        With .TextRange
            .Text = bodyText
            .Font.Name = "Arial"
            .Font.Size = 20
            .Font.Color.RGB = RGB(47, 49, 62) ' Dark Slate Gray
            .ParagraphFormat.SpaceAfter = 12
            .ParagraphFormat.LineRuleWithin = msoTrue
            .ParagraphFormat.SpaceWithin = 1.3 ' 1.3 line spacing
        End With
    End With
End Sub
